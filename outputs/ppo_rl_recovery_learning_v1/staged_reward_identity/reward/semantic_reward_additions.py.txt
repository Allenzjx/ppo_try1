RR_RETENTION_REWARD_ONLY_MODE = "rr_recapture_retention_reward_only_v1"


def _rr_retention_reward_binding(config) -> Mapping[str, Any] | None:
    """Bind the unchanged task scales once, before collection; never read hot config."""
    selected = config.path.resolve()
    expected_path = CONFIG_ROOT.parent / "ppo_rr_rl_timing_policy_learning_v1" / "reward_config.yaml"
    if selected != expected_path.resolve():
        return None  # Never silently enable this new reward in another experiment.
    if config.values.get("cooperative_preparation", {}).get("mode") != "rr_capture_cooperative_preparation_v5":
        return None
    if (config.values.get("schema") != "wlr50_clean.semantic_reward.v2"
            or config.values.get("objective_profile") != TASK_CONDITIONED_QUALITY_OBJECTIVE):
        raise ValueError("reward-only RR retention requires the frozen selected reward schema")
    from hashlib import sha256
    path = selected.with_name("stage_task_spec.yaml")
    payload = path.read_bytes()
    spec = yaml.safe_load(payload.decode("utf-8-sig"))
    expected = {
        ("schema",): "wlr50_clean.semantic_stage_task_spec.v2",
        ("potential_definition",): "global_physical_progress_v3",
        ("capture_retention_semantics",): "current_platform_region_after_placement",
        ("capture_approach_semantics",): "post_cross_FL_multiscale_positive_gap_plus_real_contact_v1",
        ("cooperative_preparation_mode",): "rr_capture_cooperative_preparation_v5",
        ("rear_owner_recovery_mode",): "issued_rear_transfer_owner_suspension_v1",
        ("p09_lift_semantics",): "functional_free_air_lift_v3",
        ("nominal", "rear_policy_timing"): "rr_rl_edge_recovery_v4",
        ("geometry", "top_gap_min_m"): -.015,
        ("geometry", "top_gap_max_m"): .025,
        ("history", "minimum_top_samples"): 2,
        ("support", "force_noise_floor_n"): .2,
    }
    for keys, value in expected.items():
        actual = spec
        for key in keys:
            actual = actual[key]
        if type(actual) is bool or actual != value:
            raise ValueError("reward-only RR retention binding differs: " + ".".join(keys))
    return {"mode": RR_RETENTION_REWARD_ONLY_MODE,
            "task_spec_sha256": sha256(payload).hexdigest(),
            "gap_scale_m": spec["geometry"]["top_gap_max_m"],
            "minimum_gap_m": spec["geometry"]["top_gap_min_m"],
            "minimum_top_samples": spec["history"]["minimum_top_samples"],
            "force_floor_n": spec["support"]["force_noise_floor_n"],
            "xy_scale_m": .25, "preparation_share_of_geometry": .25,
            "global_retention_share": .85 / 4. * .2}


def _rr_retention_reward_delta(task: Mapping[str, Any], binding: Mapping[str, Any]):
    """Replace one old retention subshare, without changing any observation or event.

    Ground/outside geometry is preparation only. Qualified AIR/EDGE can approach;
    the contact half still needs actual TOP bearing. Original legal AIR/TOP values
    are exact, including transient loss of qualification with real TOP support.
    """
    ev = task["physical_evaluator"]
    history, legs = ev["history"], ev["current_legs"]
    rr, rl = legs["RR"], legs["RL"]
    audit = {"mode": binding["mode"], "physics_tick": ev["physics_tick"],
             "scope": "unchanged", "old_retention": None, "new_retention": None,
             "delta_global_potential": 0., "changes_observation_or_contact": False}
    rl_swing = bool(rl.get("current_lift_valid") is True
        and rl.get("motion_continuation_allowed") is True and rl.get("air") is True
        and rl.get("ground_contact") is False)
    if (history["placed"]["RR"] is not True or history["placed"]["RL"] is True or rl_swing):
        return 0., audit
    if ev.get("valid") is not True or ev.get("termination_reason") is not None:
        raise ValueError("RR reward substitution requires a valid nonterminal endpoint")
    if history["active_lift"]["RR"] is not True or history["front_edge_crossed"]["RR"] is not True:
        raise ValueError("RR placed retention requires its original physical event history")
    gap = finite(rr["clearance_m"], "RR reward current gap")
    outside = finite(rr["top_xy_outside_distance_m"], "RR reward current outside distance")
    if outside < 0.:
        raise ValueError("RR reward outside distance cannot be negative")
    clip = lambda value: min(1., max(0., value))
    top = bool(rr.get("ground_contact") is False and rr.get("air") is False
        and rr.get("top_contact") is True and rr.get("top_surface_contact") is True
        and rr.get("obstacle_pair_active") is True and rr.get("within_top_xy") is True
        and rr.get("contact_surface") == "TOP")
    bearing = bool(top and rr.get("support") is True and rr.get("bearing_verified") is True
        and finite(rr["bearing_force_n"], "RR reward bearing force") >= binding["force_floor_n"])
    old_legal = bool(rr.get("within_top_xy") is True and rr.get("within_lateral_span") is True
        and rr.get("ground_contact") is False
        and (top or (rr.get("air") is True and rr.get("obstacle_pair_active") is False))
        and gap >= binding["minimum_gap_m"])
    geometry = (clip(1. - outside / binding["xy_scale_m"])
                * binding["gap_scale_m"] / (binding["gap_scale_m"] + abs(gap)))
    contact = (clip(finite(rr["consecutive_top_samples"], "RR reward TOP samples")
                    / binding["minimum_top_samples"]) if bearing and old_legal else 0.)
    old = .5 * geometry + .5 * contact if old_legal else 0.
    current_qualified = bool(rr.get("current_lift_valid") is True
        and rr.get("active_attempt") is True and rr.get("motion_continuation_allowed") is True
        and rr.get("ground_contact") is False)
    if old_legal:
        new = old  # Exact old arithmetic: no actor-visible value or accepted AIR path changes.
    else:
        fraction = 1. if current_qualified else binding["preparation_share_of_geometry"]
        new = .5 * geometry * fraction  # Outside legal TOP never earns the contact half.
    delta = binding["global_retention_share"] * (new - old)
    audit.update(scope="replace_existing_RR_retention_share", old_retention=old,
        new_retention=new, delta_global_potential=delta, geometry_proximity=geometry,
        current_qualified=current_qualified, original_legal_AIR_or_TOP=old_legal,
        current_ground=rr.get("ground_contact"), measured_TOP=top,
        measured_TOP_bearing=bearing and old_legal, contact_half=.5 * contact,
        outside_preparation_only=not old_legal and not current_qualified)
    return delta, audit


def _reward_only_potential_pair(previous, current, termination_reason, binding):
    """Decision endpoints only. Terminal zero precedes access to invalid geometry."""
    shared_before = finite(previous.task["task_progress_potential"], "potential before")
    shared_after = (None if termination_reason else
                    finite(current.task["task_progress_potential"], "potential after"))
    if binding is None:
        return shared_before, 0. if termination_reason else shared_after, None
    before_delta, before = _rr_retention_reward_delta(previous.task, binding)
    if termination_reason:
        after_delta, after = 0., {"scope": "terminal_zero_geometry_not_read"}
    else:
        after_delta, after = _rr_retention_reward_delta(current.task, binding)
    phi_before = shared_before + before_delta
    phi_after = 0. if termination_reason else shared_after + after_delta
    if not (0. <= phi_before <= 1. and 0. <= phi_after <= 1.):
        raise ValueError("reward-only replacement must remain in the existing potential range")
    return phi_before, phi_after, {
        "mode": binding["mode"], "task_spec_sha256": binding["task_spec_sha256"],
        "shared_observation_potential_before": shared_before,
        "shared_observation_potential_after": shared_after,
        "reward_potential_before": phi_before, "reward_potential_after": phi_after,
        "before": before, "after": after,
        "global_retention_share": binding["global_retention_share"],
        "not_an_additional_reward_family": True,
        "shared_task_groups_and_encoded_observation_modified": False}
